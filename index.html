Konfigurasi Database Supabase

Skema SQL Supabase (Attendance Logs)

Jalankan kode SQL berikut di **Supabase Dashboard > SQL Editor** proyek Anda untuk membuat tabel otomatis:

-- 1. Buat Tabel Attendance Logs
CREATE TABLE IF NOT EXISTS attendance_logs (
    id BIGSERIAL PRIMARY KEY,
    scanned_at TIMESTAMPTZ DEFAULT NOW(),
    no_peserta VARCHAR(20) UNIQUE NOT NULL,
    nama VARCHAR(150) NOT NULL,
    tgl_tes VARCHAR(50),
    ruang VARCHAR(50),
    waktu VARCHAR(50),
    kursi VARCHAR(10),
    note VARCHAR(255),
    raw_code TEXT,
    photo_url TEXT
);

-- 2. Aktifkan Row Level Security (RLS)
ALTER TABLE attendance_logs ENABLE ROW LEVEL SECURITY;

-- 3. Kebijakan Akses Baca & Tulis untuk Pengguna Aplikasi (Anon)
CREATE POLICY "Allow public read access" 
ON attendance_logs FOR SELECT 
USING (true);

CREATE POLICY "Allow public insert access" 
ON attendance_logs FOR INSERT 
WITH CHECK (true);

-- 4. Indeks untuk Pencarian Cepat
CREATE INDEX IF NOT EXISTS idx_attendance_no_peserta ON attendance_logs(no_peserta);
CREATE INDEX IF NOT EXISTS idx_attendance_scanned_at ON attendance_logs(scanned_at DESC);
